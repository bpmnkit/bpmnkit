document_verified >(verification-passed: verified = true) screening[service Sanctions & PEP Screening | job=io.camunda:http-json:1 in.authentication.type="bearer" in.authentication.token=secrets.SCREENING_SERVICE_TOKEN in.method="POST" in.url=screeningServiceUrl + "/screen" in.body={ fullName: applicant.fullName, dateOfBirth: applicant.dateOfBirth, nationality: applicant.nationality } in.connectionTimeoutInSeconds="20" in.readTimeoutInSeconds="20"] > due_diligence_level[xor Due Diligence Level?]
document_verified >(verification-failed: default) increment_retry_count[script Increment Retry Count | result=retryCount feel=if is defined(retryCount) then retryCount + 1 else 1] > retries_remaining[xor Retries Remaining?]
due_diligence_level >(standard-dd: sanctionsHit = false and pep = false) standard_due_diligence[service Standard Due Diligence | job=due-diligence-service] > xor[xor] > final_risk_classification[rule Final Risk Classification] > classification_outcome[xor Classification Outcome?]
due_diligence_level >(enhanced-dd: default) enhanced_due_diligence_review[user Enhanced Due Diligence Review | form=enhanced-dd-review-form] > xor
retries_remaining >(retry-upload: retryCount <= 2) upload_identity_document
retries_remaining >(retries-exhausted: default) send_rejection_notice[service Send Rejection Notice | job=email-sender] > rejected_verification_failed[end Rejected — Verification Failed]
classification_outcome >(approved: riskClassification.decision = "approve") activate_customer_account[service Activate Customer Account | job=account-activation-service in.customerId=applicant.id] > customer_approved[end Customer Approved]
classification_outcome >(escalated: riskClassification.decision = "escalate") escalate_to_compliance[service Escalate to Compliance | job=compliance-escalation-service] > escalated_to_compliance[end Escalated to Compliance]
