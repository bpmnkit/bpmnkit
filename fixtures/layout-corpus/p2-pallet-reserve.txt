carrierAnswer > silent[catch:timer 1 day silent | after=P1D] > releasePallet[service Release pallet | job=release-pallet] > [end]
carrierAnswer > declined[catch Carrier declined | msg=carrier-declined key=loadId] > rebookLoad[user Rebook with other carrier] > [end]

confirmFailed[boundary:error Confirm failed | on=confirmLoad error=CONFIRM_FAILED] > callCarrier[user Call carrier] > [end]
